			\hphantom{\qquad \displaystyle \frac{(4-3i)}{(3+2i)}}\quad = \frac{12-17i+6i^2}{9- 4i^2} \quad = \frac{12-17i-6}{9+ 4}\\[1em]
			\hphantom{\qquad \displaystyle \frac{(4-3i)}{(3+2i)}}\quad = \frac{6-17i}{13} \quad = \frac{6}{13} - \frac{17}{13}i$
\end{enumerate}
\end{itembox}
