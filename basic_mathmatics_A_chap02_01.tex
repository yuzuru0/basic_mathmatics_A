\section{方程式}
\subsection{2次方程式}

{\Large
\begin{itemize}
	\item 方程式 … 等式の文字に特別な値を入れると成り立つ式\\[0em]
	\item $x$の方程式 … 等式の中に文字$x$を含む方程式\\[0em]
	\item 解 … 方程式を満たす$x$の値\\[0em]
	\item 未知数 … $x$の方程式の$x$のこと\\[0em]
	\item 2次方程式 … 最大次数2の$ ax^2+bx+c=0$の方程式\\[0em]
\end{itemize}

\begin{itembox}[l]{例題1: 方程式$6x^2-x-2=0$の解}
\quad \\[.5em]
因数分解して
\vspace{-1em}
\begin{align*}
(3x-2)(2x+1) = 0
\end{align*}
%
\quad \\[-.5em]
成り立つためには、どちらかの括弧内が0になれば良い
\begin{itemize}
	\item $3x-2 =0$
	\item $2x+1 =0$
\end{itemize}

つまり、
\begin{align*}
x = \frac{2}{3} \quad \text{または}\quad x=-\frac{1}{2}
\end{align*}
\end{itembox}

}

{\Large
\begin{itembox}[l]{2次方程式の解の公式}
\begin{align*}
x = \frac{-b \pm \sqrt{b^2 -4ac}}{2a}
\end{align*}
\end{itembox}

【導出】
\begin{align*}
&ax^2+bx+c\ =\ 0	\hspace{10em}																	&& \\[.5em]
&x^2 + \frac{b}{a}x + \frac{c}{a}\ =\ 0																&& {\normalsize \leftarrow \text{両辺を$a$で割る}}\\[1em]
&x^2+\frac{b}{a}x \ =\  -\frac{c}{a}																	&& {\normalsize \leftarrow \text{$\frac{c}{a}$の移項}}\\[1em]
&x^2+\frac{b}{a}x +\left( \frac{b}{2a} \right)^2 \ =\  \left( \frac{b}{2a} \right)^2 -\frac{c}{a}	&& {\normalsize \leftarrow \text{両辺に$\left( \frac{b}{2a} \right)^2$追加}}\\[1em]
&\left( x+\frac{b}{2a} \right)^2 \ =\  \frac{b^2 -4ac}{4a^2}										&& {\normalsize \leftarrow \text{両辺整理}}\\[1em]
& x+\frac{b}{2a} \ =\  \pm \frac{\sqrt{b^2 -4ac}}{\sqrt{4a^2}}										&& {\normalsize \leftarrow \text{両辺の平方根}}\\[1em]
& x \ =\ -\frac{b}{2a} \pm \frac{\sqrt{b^2 -4ac}}{2a}												&& {\normalsize \leftarrow \text{$\frac{b}{2a}$の移項}}\\[1em]
& x \ =\ \frac{-b \pm \sqrt{b^2 -4ac}}{2a}															&& {\normalsize \leftarrow \text{整理}}\\[1em]
\end{align*}

根号の中の式$b^2-4ac$の値によって解の形式が変わる。\\
$D=b^2-4ac$を\textbf{判別式}と呼び、方程式の解の形式を判定する。

}

{\Large
\begin{itembox}[l]{判別式と解の関係}
2次方程式\ $ax^2+bx+c=0$\ の解は解の公式により
\begin{align*}
x =\frac{-b\pm \sqrt{D}}{2a}
\end{align*}
であり判別式$D=b^2-4ac$の値で解が変わる。
\quad\\[-.5em]
\begin{enumerate}
\renewcommand{\theenumi}{\Roman{enumi}}	% 数字の形式を (i) に変更
\renewcommand{\labelenumi}{(\theenumi)}% ラベルの表示形式からドットを除去してカッコを付ける
%
	\item \quad $D>0$ならば、2つの実数解
	\item \quad $D=0$ならば、2重解
	\item \quad $D<0$ならば、2つの虚数解
\end{enumerate}
\end{itembox}
\quad \\[-.5em]

\begin{enumerate}
\renewcommand{\theenumi}{\Roman{enumi}}	% 数字の形式を (i) に変更
\renewcommand{\labelenumi}{(\theenumi)}% ラベルの表示形式からドットを除去してカッコを付ける
\setcounter{enumi}{0}
%
	\item $D=b^2-4ac >0$の場合
\end{enumerate}
\quad\\[-1.5em]

$\sqrt{b^2-4ac}$の値は正の実数になる。\\よって方程式は\textbf{異なる2つの実数解}をもつ\\[1em]

【例1】$2x^2+5x+1=0$の解\\

\begin{align*}
x = \frac{-5 \pm \sqrt{5^2 - 4 \cdot 2 \cdot 1}}{2\cdot 2} \quad =\frac{-5 \pm \sqrt{25-8}}{4} \quad = \frac{-5\pm \sqrt{17}}{4}
\end{align*}

}
\newpage
%
%
{\Large
\begin{enumerate}
\renewcommand{\theenumi}{\Roman{enumi}}	% 数字の形式を (i) に変更
\renewcommand{\labelenumi}{(\theenumi)}% ラベルの表示形式からドットを除去してカッコを付ける
\setcounter{enumi}{1}
%
	\item $D=b^2-4ac =0$の場合
\end{enumerate}
\quad \\[-3em]

$\sqrt{b^2-4ac}=0$なので解は$x=-\frac{b}{2a}$と\textbf{1つの実数解}となる。

1つの解には2つの実数解が重複するとみて\textbf{2重解}と呼ぶ。
\\[1em]

【例2】$x^2-8x+16=0$の解\\
\begin{align*}
x &= \frac{-(-8) \pm \sqrt{(-8)^2 -4 \cdot 1 \cdot 16}}{2 \cdot 1} \quad = \frac{8\pm\sqrt{64-64}}{2} \\[1em]
&= \frac{8\pm0}{2} = 4
\end{align*}
\quad \\[1em]
%
%
%
\begin{enumerate}
\renewcommand{\theenumi}{\Roman{enumi}}	% 数字の形式を (i) に変更
\renewcommand{\labelenumi}{(\theenumi)}% ラベルの表示形式からドットを除去してカッコを付ける
\setcounter{enumi}{2}
%
	\item $D=b^2-4ac <0$の場合
\end{enumerate}
\quad \\[-3em]

$\sqrt{b^2-4ac}$の値は虚数になる。\\
よって方程式は\textbf{異なる2つの虚数解}をもつ\\[1em]

【例3】$2x^2-3x+5=0$の解\\
\begin{align*}
x=\frac{3 \pm \sqrt{9-40}}{4} \quad =\frac{3 \pm \sqrt{-31}}{4}\quad =\frac{3 \pm \sqrt{31}i}{4}
\end{align*}

}

{\Large
\begin{itembox}[l]{例題2: 2重解を持つようにkを決める}
\quad \\[.5em]
$x^2+(k-3)x+k=0$が2重解を持つように$k$を設定\\[.5em]
$\rightarrow$判別式$D=0$になれば良い。
\begin{align*}
D &= (k-3)^2 -4 \cdot 1 \cdot k \quad = k^2-6k+9 -4k \\[.5em]
  &= k^2-10k+9 \quad = (k-1)(k-9) = 0
\end{align*}
%
よって、$k=1, 9$\\

このときの解は解の公式から
\begin{align*}
\frac{-(k-3) \pm \overbrace{\sqrt{(k-3)^2-4\cdot 1 \cdot k}}^{0}}{2} = -\frac{k-3}{2}
\end{align*}

つまり
\begin{itemize}
	\item $k=1$のとき$-\frac{1-3}{2} = +1$
	\item $k=9$のとき$-\frac{9-3}{2}=-3$
\end{itemize}

\end{itembox}

}
