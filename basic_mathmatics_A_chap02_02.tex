{\Large
	\begin{itemize}
		\item 不等式 … 不等号$>,<,\geqq, \leqq $を用いて大小を表した式 \footnote{虚数は考えない}\\[-.5em]
	\end{itemize}
	\begin{itembox}[l]{不等式の性質(1)}
		\begin{enumerate}
			\renewcommand{\theenumi}{\Roman{enumi}}	% 数字の形式を (i) に変更
			\renewcommand{\labelenumi}{(\theenumi)}% ラベルの表示形式からドットを除去してカッコを付ける

			\item \quad $a>b$ならば\quad $a-b>0$\\
					\quad $a-b>0$ ならば \quad $a>b$ \label{lst:inequality_01}

	\end{enumerate}
	\end{itembox}

	\subsubsection{1次不等式の解法}
	\begin{itemize}
		\item 不等式の解 … 与えられた不等式の成り立つ$x$の範囲。\\[-.0em]
				\hphantom{不等式の解 … }$x<3$\  などと表す。\\[-.5em]
		\item 不等式を解く … 不等式の解を求めること。\\[-.5em]
		\item 1次不等式 … 右辺が$0$のときの左辺が1次の不等式。\\
				\hphantom{1次不等式 … }$x-4>0$ \  のような式。
	\end{itemize}
%
%
	\quad \\[-2.5em]
	\begin{itembox}[l]{例題1: 1次不等式の解}
	\vspace{-1em}
		\begin{align*}
			\frac{x+3}{2} > 2x-3
		\end{align*}
		%
%		\quad\\
		\begin{flalign*}
		&&	x+3 &> 4x-6		&& \leftarrow \text{両辺に}2\text{をかける}\\
%		&&	x+3 -4x &> 4x -4x -6	&& 
		&&	-3x+3 &> -6		&&	\leftarrow \text{両辺から}4x\text{を引く}\\
		&&	-3x &> -9		&& \leftarrow \text{両辺から}3\text{を引く}\\
		&&	x &< 3			&& \leftarrow \text{両辺を}-3\text{で割る}
		\end{flalign*}
		よって解は\quad $x <3$\quad である。
	\end{itembox}
}

\subsubsection{不等式の証明}

不等式の証明は、性質で説明した(I)から(VI)を使用する\\[-1.5em]

	\begin{itembox}[l]{例題6: 1次不等式の証明1}
%		\vspace{-0em}
		\quad \\[-1.em]
		$x\geqq 1,\ y\geqq 1$のとき、$xy+1 \geqq x+y$が成り立つことを証明せよ.また、等号の成り立つ条件を調べよ。
		\\[-0.5em]
		
		与式を変形すると
		\begin{flalign*}
			&& xy+1 \ &\geqq \ x+y				&&	\\
			&& (xy+1) - (x+y)\  & \geqq \  0	&& \leftarrow (I)	\\
			&& xy -x -y +1 \  &\geqq \ 0		&&					\\
			&& x(y -1) -(y -1) \  &\geqq \ 0	&&					\\
			&& (x-1)(y -1) \  &\geqq \ 0		&&					
		\end{flalign*}
		\quad \\[-2.5em]である。
		
		与えられた条件$x\geqq 1,\ y\geqq 1$は(I)により
		\begin{align*}
			x -1 \geqq 0, \qquad y-1 \geqq 0
		\end{align*}
		\quad \\[-2.5em]である。
		
		つまり与式のカッコ内はどちらも常に正または0であり、\\
		$xy+1 \geqq x+y$は成り立つことが証明される。\\[0em]
		
		等号が成り立つ条件は、$(x-1)(y -1) \  =\  0$のときであり、どちらかのカッコ内が0である。\\
		そのため$x-1=0,\ y-1=0$となる。
		
		つまり、$x=1$または$y=1$のときに等号は成り立つ。
	\end{itembox}
}

		\begin{itembox}[l]{例題7: 1次不等式の証明2}
%		\vspace{-0em}
		\quad \\[-0.em]
		$a>b>0, \quad c>d>0$のとき、$ac>bd$を証明せよ。
		\\[-0.5em]
		\begin{flalign*}
			&& &a-b >0 							&& \leftarrow (I)	\\
			&& &c>0なので\quad c(a-b) = ac-bc>0	&& \leftarrow (V)	\\
		\end{flalign*}
		\\[-5em]
		\begin{flalign*}
			&& &c-d >0 							&& \leftarrow (I)	\\
			&& &b>0なので\quad b(c-d) = bc-bd>0	&& \leftarrow (V)	\\
		\end{flalign*}
		$ac-bc>0$\ と\ $bc-bd>0$\ を足すことで、
		\begin{align*}
		ac -bd &>0	\\
		ac >bd
		\end{align*}
	\end{itembox}
	\\[2em]
	不等式の証明には、下記実数の性質も良く用いる。\\[-1em]
	\begin{itembox}[l]{実数の性質}
		任意の実数$a$は、\quad $a^2 \geqq0$\quad が成り立つ
	\end{itembox}
	\\[1em]
	\noindent【例】上記より$(a-b)^2 \geqq 0$である。
	展開すると
	\begin{align*}
		a^2+b^2-2ab &\geqq 0\\
		a^2 +b^2 &\geqq 2ab
	\end{align*}
	となる。\\[.5em]
	等号は$(a-b)^2=0$なので、$a-b=0$つまり$a=b$のとき成り立つ
